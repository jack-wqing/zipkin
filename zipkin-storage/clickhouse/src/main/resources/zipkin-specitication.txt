CREATE TABLE log.zipkin_spans
(
    `traceId` String,
    `parentId` String,
    `id` String,
    `kind` String,
    `name` String,
    `duration` UInt32,
    `timestamp` UInt64,
    `timestampMillis` UInt64,
    `time` DateTime,
    `serviceName` String,
    `localEndpoint` Map(String, String),
    `remoteEndpoint` Map(String, String),
    `annotations` Map(String, String),
    `tags` Map(String, String)
)
ENGINE = MergeTree
PARTITION BY toYYYYMMDD(time)
ORDER BY (time, serviceName)
TTL time + toIntervalDay(7)
SETTINGS index_granularity = 8192;
